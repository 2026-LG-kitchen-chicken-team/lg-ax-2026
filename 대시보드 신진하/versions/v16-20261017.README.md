```powershell
git clone --branch codex/dashboard-shin-jinha-v16-classic-records-20261017 https://github.com/2026-LG-kitchen-chicken-team/lg-ax-2026.git
cd "lg-ax-2026/대시보드 신진하/versions/v16-20261017"
npm ci
npm run dev
```
