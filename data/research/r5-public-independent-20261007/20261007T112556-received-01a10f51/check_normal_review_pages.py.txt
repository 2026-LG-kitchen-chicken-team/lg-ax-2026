def normalize(t):return " ".join(unicodedata.normalize("NFKC",t).split())
def sha(b):return hashlib.sha256(b).hexdigest()
class Node:
    def __init__(self,tag,attrs,parent=None):self.tag=tag;self.attrs=dict(attrs);self.parent=parent;self.children=[]
    def get(self,k):return self.attrs.get(k)
    def all(self):
        for child in self.children:
            if isinstance(child,Node):yield child;yield from child.all()
    def select(self,selector):
        def matches(n):
            if selector.startswith('#'):return n.get('id')==selector[1:]
            if selector.startswith('.'):return selector[1:] in (n.get('class') or '').split()
            if selector=='li[data-review-id]':return n.tag=='li' and n.get('data-review-id') is not None
            if selector=='[class*=date]':return 'date' in (n.get('class') or '')
            return False
        return [n for n in self.all() if matches(n)]
    def select_one(self,selector):return next(iter(self.select(selector)),None)
    def get_text(self,sep=' ',strip=False):
        parts=[]
        for child in self.children:
            if isinstance(child,str):parts.append(child.strip() if strip else child)
            elif child.tag not in ('script','style'):parts.append(child.get_text(sep,strip))
        return sep.join(p for p in parts if p)
class Tree(HTMLParser):
    def __init__(self):super().__init__(convert_charrefs=True);self.root=Node('root',[]);self.current=self.root
    def handle_starttag(self,tag,attrs):
        n=Node(tag,attrs,self.current);self.current.children.append(n)
        if tag not in ('br','hr','img','input','meta','link','area','base','embed','param','source','track','wbr'):self.current=n
    def handle_endtag(self,tag):
        n=self.current
        while n.parent:
            if n.tag==tag:self.current=n.parent;return
            n=n.parent
    def handle_data(self,data):self.current.children.append(data)
