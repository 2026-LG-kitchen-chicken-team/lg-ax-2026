def digest(raw):
    return hashlib.sha256(raw).hexdigest()


def read(relative):
    raw = (ROOT / relative).read_bytes()
    PINS.append({'commit': BASE, 'path': relative, 'bytes': len(raw), 'sha256': digest(raw)})
    return json.loads(raw) if relative.endswith('.json') else raw


def pr270(name):
    index = json.loads((RUN / 'intake270/index.json').read_text(encoding='utf-8'))
    row = next(x for x in index if x['path'].endswith('/' + name))
    raw = (RUN / 'intake270' / row['local_file']).read_bytes()
    PINS.append({'commit': row['head'], 'path': row['path'], 'bytes': len(raw), 'sha256': digest(raw), 'merged': False})
    return json.loads(raw)


def check(label, passed, source_id=None):
    row = {'check': label, 'passed': bool(passed)}
    if source_id:
        row['source_id'] = source_id
    CHECKS.append(row)
    if not passed:
        ERRORS.append(row)
    return bool(passed)


def write(name, obj):
    (OUT / name).write_text(json.dumps(obj, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')


def write_csv(name, rows):
    with (OUT / name).open('w', encoding='utf-8', newline='') as stream:
        writer = csv.DictWriter(stream, fieldnames=list(rows[0]))
        writer.writeheader()
        writer.writerows(rows)


def normalized(text):
    return ' '.join(unicodedata.normalize('NFKC', text).split())


class VisibleText(HTMLParser):
    def __init__(self):
        super().__init__()
        self.parts = []
        self.ignore = 0

    def handle_starttag(self, tag, attrs):
        if tag in ('script', 'style'):
            self.ignore += 1

    def handle_endtag(self, tag):
        if tag in ('script', 'style') and self.ignore:
            self.ignore -= 1
        if tag in ('div', 'p', 'td', 'th', 'li', 'tr'):
            self.parts.append('\n')

    def handle_data(self, value):
        if not self.ignore:
            self.parts.append(value)


def visible(text):
    parser = VisibleText()
    parser.feed(text)
    return html.unescape(''.join(parser.parts)).strip()
