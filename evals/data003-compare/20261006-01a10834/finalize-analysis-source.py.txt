def read(path):
    return json.loads(path.read_text("utf-8-sig"))


def save(name, data):
    with (PREP / name).open("x", encoding="utf-8", newline="\n") as handle:
        json.dump(data, handle, ensure_ascii=False, indent=2)
        handle.write("\n")


def csvwrite(name, rows):
    with (PREP / name).open("x", encoding="utf-8-sig", newline="") as handle:
        writer = csv.DictWriter(handle, fieldnames=list(rows[0]))
        writer.writeheader()
        writer.writerows(rows)


queries = {q["id"]: q for q in read(OLD / "evals/data003-preparation/20261005/queries.json")}
results = read(OLD / "evals/data003-live/20261005/retrieval-v1/results.json")
checks = []
for row in results["results"]:
    gold = set(queries[row["query_id"]]["relevant_evidence_ids"])
    for attempt in row["latency_attempts"]:
        if not gold:
            continue
        ids = [e["id"] for e in attempt.get("ranking", [])[:3]]
        values = {"hit_at_1": int(bool(ids) and ids[0] in gold),
                  "recall_at_3": len(gold.intersection(ids)) / len(gold),
                  "mrr_at_3": next((1 / (i + 1) for i, eid in enumerate(ids) if eid in gold), 0)}
        checks.extend({"name": f'{row["method"]}:{row["query_id"]}:{attempt["repeat"]}:{k}',
                       "pass": math.isclose(v, row[k], abs_tol=1e-10)} for k, v in values.items())

ledger = read(PREP / "ocr-error-ledger.json")
numeric_rechecks = []
for row in ledger["rows"]:
    if not row["field_target"]:
        continue
    text = " ".join(row["recognized"].split())
    if "*" in row["gold_text"]:
        text = re.sub(r"\*\d+$", "", text)
    observed = [float(n) for n in re.findall(r"(?<![\w.])-?\d+(?:\.\d+)?", text)]
    value = row["field_target"]["value"]
    target = value if isinstance(value, list) else [value]
    correct = sum(i < len(observed) and observed[i] == n for i, n in enumerate(target))
    units = re.findall(r"[A-Za-z]+|인용", text)
    unit_correct = min(sum(u == row["field_target"]["unit"] for u in units), row["unit_total"]) if row["unit_total"] else None
    checks.extend([{"name": row["region_id"] + ":numeric_value", "pass": correct == row["numeric_correct"]},
                   {"name": row["region_id"] + ":strict_numeric_field", "pass": (observed == target) == row["strict_numeric_field_correct"]},
                   {"name": row["region_id"] + ":explicit_unit", "pass": unit_correct == row["explicit_unit_correct"]}])
    numeric_rechecks.append({"region_id": row["region_id"], "source_gold": row["gold_text"], "recognized": row["recognized"],
        "target_values": target, "parsed_values": observed, "parsed_units": units, "base_numeric_correct": correct,
        "strict_numeric_field_correct": observed == target, "explicit_unit_correct": unit_correct,
        "footnote_error": "ASCII *1 recognized as Unicode ∗1, so strict parser retains an extra 1" if row["region_id"] == "OCR-MW-TABLE-05" else None})
assert all(c["pass"] for c in checks)
save("independent-arithmetic-checks.json", {"kind": "saved ranking/typed-field recalculation, not independent Agent evaluation",
    "checks": checks, "pass_count": len(checks), "new_model_calls": 0, "new_OCR_calls": 0,
    "numeric_fields": numeric_rechecks})
