def safe_check_value(value):
    if isinstance(value,(dict,list)):
        return {"value_kind":type(value).__name__,"item_count":len(value),"canonical_json_sha256":digest(canonical(value))}
    if isinstance(value,str) and len(value)>600:

    return value

def cited(value):
    result=set()
    if isinstance(value,dict):

    elif isinstance(value,list):
        for v in value:result.update(cited(v))
    return result
