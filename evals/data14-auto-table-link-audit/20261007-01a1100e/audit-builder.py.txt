def read(path):
    return json.loads((ROOT / path).read_bytes())


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def write_text(name, text):
    target = (OUT / name).resolve()
    if not target.is_relative_to(OUT):
        raise ValueError("Audit output escaped unique directory")
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes(text.replace("\r\n", "\n").encode("utf-8"))


def write_json(name, value):
    write_text(name, json.dumps(value, ensure_ascii=False, indent=2) + "\n")


def write_csv(name, rows):
    stream = io.StringIO(newline="")
    writer = csv.DictWriter(stream, fieldnames=list(rows[0]), lineterminator="\n")
    writer.writeheader()
    for row in rows:
        writer.writerow({k: json.dumps(v, ensure_ascii=False) if isinstance(v, (dict, list)) else v for k, v in row.items()})
    write_text(name, stream.getvalue())


def csv_rows(path):
    with (ROOT / path).open(encoding="utf-8-sig", newline="") as stream:
        return list(csv.DictReader(stream))


def check(name, actual, expected):
    okay = actual == expected
    checks.append({"check": name, "actual": actual, "expected": expected, "passed": okay})
    if not okay:
        write_json("verification-failure.json", checks)
        raise ValueError(name)


def git(*args):
    return subprocess.check_output(["git", "-c", "core.quotepath=false", "-c", "core.fsmonitor=false", *args], cwd=ROOT).decode("utf-8").strip()


def norm(text):
    return " ".join(unicodedata.normalize("NFC", text).split())


def edit_distance(a, b):
    previous = list(range(len(b) + 1))
    for i, left in enumerate(a, 1):
        current = [i]
        for j, right in enumerate(b, 1):
            current.append(min(current[-1] + 1, previous[j] + 1, previous[j - 1] + (left != right)))
        previous = current
    return previous[-1]

def key_inventory(value):
    keys = set()
    if isinstance(value, dict):

            keys.update(key_inventory(child))
    elif isinstance(value, list):
        for child in value:
            keys.update(key_inventory(child))
    return keys

def esc(value):
    if isinstance(value, (dict, list)):
        value = json.dumps(value, ensure_ascii=False)
    return str(value).replace("|", "\\|").replace("*", "\\*").replace("\n", " ")


def md_table(headers, data):
    return "| " + " | ".join(headers) + " |\n| " + " | ".join(["---"] * len(headers)) + " |\n" + "\n".join("| " + " | ".join(esc(v) for v in row) + " |" for row in data)

## 2. 저장 OCR 문자 대조 — 14영역

{region_table}

원문 위치/gold는 14영역 모두 수동 확인했다. 공백 차이 5영역과 `*`(U+002A)→`∗`(U+2217) 1영역을 문자 오류로 보존했다. 공백 판정은 source gold/native text층의 문자 기록을 함께 사용하며 그림에서 코드포인트를 추정하지 않는다. 자세한 bbox·개정·SHA·JSON pointer·단위 위치는 [recognition-review.csv](recognition-review.csv)에 있다.

`OCR-MW-TABLE-05`의 **1은 원문 각주 번호**다. 저장 evaluator는 ASCII `*` 뒤 숫자만 제거하므로 OCR의 `∗1`을 제거하지 못하고 `[1000,1]`로 기록했다. 이는 각주/본값 분리 실패이며 OCR이 새 본값 1을 만들어낸 사례로 표현하지 않았다. 기존 점수 10/11은 유지했다.
